### 3.1 风格包 `style-packs.js`

每套包字段（全部为 prompt 就绪的中文字符串，不要写名词清单）：

```js
{
  id: "S1", name: "白底极简",
  colorTone: "…",            // 色温/色调
  lighting: "…",             // 布光方案（替换写死光线行的核心）
  compositionFamily: "…",    // 构图族
  background: "…",           // 背景策略
  fontMood: "…",             // 字体气质
  avoid: ["…", "…"],         // 禁忌（数组）
  lockLine: "…",             // 一句话摘要，供【视觉风格锁】行使用
  variantBias: { hero: ["A"], materialDetail: ["A"] }  // 可选：模块变体偏好排序
}
```

### 3.2 构图变体 `composition-variants.js`

结构：`MODULE_VARIANTS = { <moduleId>: [ { id, letter, name, composition, firstLook, eyeFlow, textSafeZone, avoid, lighting? }, ... ] }`。
`id` 形如 `hero-A`；`letter` 为 `"A"|"B"|"C"|"D"`；`lighting` 仅在有变体专属布光时给（可省略）。
moduleId 必须与 v1 `SCREEN_MODULES` 的 9 个 id 完全一致：`hero / painReverse / coreSellingPoint / materialDetail / useScenario / specInfo / trustProof / conversionClose / fitStructure`。

## 4. 函数规格（`prompts/server/detail-skills-v2/index.js`）

### 4.1 `resolveDetailSkillVersion(input)` —— 版本裁决

```js
export function resolveDetailSkillVersion(input = {}) {
  const requested = String(input?.detailSkillVersion || "").trim().toLowerCase();
  if (requested === "v1" || requested === "v2") return requested;
  const fromEnv = String(process.env.JINGYIN_DETAIL_SKILL_VERSION || "").trim().toLowerCase();
  return fromEnv === "v2" ? "v2" : "v1";
}
```

- 优先级：请求级 `detailSkillVersion` > env `JINGYIN_DETAIL_SKILL_VERSION` > **默认 v1**。
- env 在**函数体内**读取（不得模块顶层求值、不得进 `runtime-config.js`）。
- `detail-ai.js` 与 `detail-middleware.js` 用同一函数、同一 input ⇒ 两处裁决必然一致。

### 4.2 `stylePackForGroup({ groupIndex, platformSkill, userChoice })`

```js
export function stylePackForGroup({ groupIndex = 1, platformSkill, userChoice = "auto" }) {
  const index = Math.max(1, Number.parseInt(groupIndex, 10) || 1);
  const pool = autoPoolIds();                       // S2..S8
  const choice = String(userChoice || "auto").trim();
  if (choice && choice !== "auto" && STYLE_PACKS[choice] && allowedPackIds(platformSkill).includes(choice)) {
    return { pack: STYLE_PACKS[choice], groupIndex: index, reason: `手动指定 ${choice}《${STYLE_PACKS[choice].name}》` };
  }
  const picked = pool[(index - 1) % pool.length];
  const pack = STYLE_PACKS[picked];
  const knownButBanned = choice && choice !== "auto" && STYLE_PACKS[choice];
  return {
    pack,
    groupIndex: index,
    reason: knownButBanned
      ? `所选 ${choice} 不适用${platformSkill?.platform || ""}${platformSkill?.pageType || ""}规范，已按第 ${index} 组自动轮换为 ${picked}《${pack.name}》`
      : `第 ${index} 组自动轮换：${picked}《${pack.name}》`
  };
}
```

- **组间分化**：连续两组必不同（pool 长 7，轮转取模）。
- **无效 id / 平台禁用** → 落回自动轮换并写 reason（不报错、不阻断）。
- `groupIndex` 来源：前端 `detailGroupIndex = detailGroups.length + 1`；服务端缺省 1（旧前端/预设不传也能跑，只是恒 S2）。

### 4.3 `variantFor(moduleId, screenIndex, stylePack, groupIndex, aiChoice)`

```js
export function variantFor(moduleId, screenIndex, stylePack, groupIndex, aiChoice) {
  const variants = MODULE_VARIANTS[moduleId] || MODULE_VARIANTS.coreSellingPoint;
  const choice = String(aiChoice || "").trim();
  if (choice) {
    const hit = variants.find((v) => v.id === choice || v.letter === choice);
    if (hit) return hit;                      // AI 选择有效 → 采纳
  }
  const bias = stylePack?.variantBias?.[moduleId];
  let ordered = variants;
  if (Array.isArray(bias) && bias.length > 0) {
    const preferred = bias.map((letter) => variants.find((v) => v.letter === letter)).filter(Boolean);
    ordered = [...preferred, ...variants.filter((v) => !bias.includes(v.letter))];
  }
  const offset = (Math.max(1, Number.parseInt(groupIndex, 10) || 1) - 1 + Math.max(0, Number.parseInt(screenIndex, 10) || 0)) % ordered.length;
  return ordered[offset];
}
```

- 组内：相邻屏 offset+1 ⇒ 不连续重复同一变体；跨组：起始 offset 随 groupIndex 平移 ⇒ 同模块变体不同。
- AI 的 `compositionVariantId` 只接受**该模块存在**的 id/字母，否则按规则分配。

## 11. 回收站 `.lock` 清理纪律（硬性·2026-10-16 根因修订版）

> 旧版「扫工作区 `.lock` 报 0」**已废止**——`.lock` 不在工作区产生，该纪律恒为 0、形同虚设，回收站照涨（实测 12273 → 15496）。

**根因（已取证）**：safe-delete / 软删除类工具批量删除时，在 `%TEMP%\codebuddy-safe-delete-bulk\<hash>\` 自建作业锁目录，作业结束又把整个 temp 作业目录丢进回收站 → `.lock` 以 `$R*.lock` 形态堆进 `C:\$Recycle.Bin\...-1001`。抽样 400 个索引 **400/400 来自该目录**。**谁用 safe-delete 谁造垃圾。**

**新三律（执行者必守）**：

1. 全程**禁用** safe-delete / 移入回收站 / 任何「先暂存再删」封装；删除只用 PowerShell 原生 `Remove-Item -Recurse -Force` 或 node `fs.rmSync(p, { recursive: true, force: true })`。
2. 收尾清两处并各报数：`%TEMP%\codebuddy-safe-delete*` 残留 → **0**；回收站 `.lock` 计数 → **0**（取证手法：回收站 `$I<id>.<ext>` 索引含原路径，`version(8) size(8) filetime(8) pathLen(4) path(UTF16LE)`，pathLen 在偏移 **24**、路径从偏移 **28** 起，Win10 v2 格式）。
3. 回报附**回收站 `.lock` 计数**，不得高于执行前基线。

**并行会话警告**：另有 AI 会话并行运行时 `.lock` 会以 ~100/分钟再生——清理与计数须挑**无并行会话窗口**做，否则边清边涨（实测 33 分钟 PowerShell 清理反从 15496 涨到 21103）。批量清理用 node `fs.rmSync`（比 `Remove-Item` 管道快一个量级）。根治还须系统层设置（回收站属性勾「不将文件移到回收站」或设 1–2 GB 自定义大小上限），属执行单覆盖范围外，由用户侧配置。
